local function CreateMiniReset()

    if game.CoreGui:FindFirstChild("MiniResetGUI") then

        game.CoreGui.MiniResetGUI:Destroy()

    end

    local gui = Instance.new("ScreenGui", game.CoreGui)

    gui.Name = "MiniResetGUI"

    local btn = Instance.new("TextButton", gui)

    btn.Size = UDim2.new(0, 70, 0, 25)

    btn.Position = UDim2.new(0, 6, 0, 6)

    btn.Text = "RESET"

    btn.BackgroundColor3 = Color3.fromRGB(140, 30, 30)

    btn.TextColor3 = Color3.fromRGB(255,255,255)

    btn.Font = Enum.Font.GothamBold

    btn.TextSize = 12

    btn.BorderSizePixel = 0

    Instance.new("UICorner", btn).CornerRadius = UDim.new(0,6)

    btn.MouseButton1Click:Connect(function()

        if isfile(SaveFile) then

            delfile(SaveFile)

        end

        gui:Destroy()

        task.wait(0.2)

        warn("[SAFE] Reset xong, hãy inject lại script selector.")

    end)

end

CreateMiniReset()
